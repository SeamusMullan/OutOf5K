% === CRITICAL FIX: Disable soul's text case transformations ===
% Must be done AFTER \documentclass loads the Tufte class
% The Tufte class uses textcase package which relies on soul for text transformations
% Soul breaks on complex content, so we replace it with simpler TeX primitives
\makeatletter
\DeclareRobustCommand\MakeTextUppercase{%
  \def\NoCaseChange##1{\noexpand\NoCaseChange{\noexpand##1}}%
  \@uclcnotmath{\uppercase}{\def\NoCaseChange##1{##1}}%
}
\DeclareRobustCommand\MakeTextLowercase{%
  \def\NoCaseChange##1{\noexpand\NoCaseChange{\noexpand##1}}%
  \@uclcnotmath{\lowercase}{\def\NoCaseChange##1{##1}}%
}
\makeatother
